% macros/abbreviations.tex — Latin abbreviations and common shortcuts.
%
% These wrap a period + correct spacing so prose stays readable.
% Use \xspace at the end so a following non-space character also gets a space.

% Latin
\newcommand{\eg}{e.g.\@\xspace}
\newcommand{\ie}{i.e.\@\xspace}
\newcommand{\cf}{cf.\@\xspace}
\newcommand{\etal}{et al.\@\xspace}
\newcommand{\etc}{etc.\@\xspace}
\newcommand{\vs}{vs.\@\xspace}
\newcommand{\wrt}{w.r.t.\@\xspace}
\newcommand{\aka}{a.k.a.\@\xspace}
\newcommand{\viz}{viz.\@\xspace}

% Academic
\newcommand{\sic}{[\emph{sic}]\xspace}
\newcommand{\ibid}{ibid.\@\xspace}
\newcommand{\opcit}{op.\@\,cit.\@\xspace}

% Typographic dashes (use sparingly; – and — also work directly)
\providecommand{\emdash}{\textemdash\xspace}
\providecommand{\ndash}{\textendash\xspace}

% Units / numbers
\newcommand{\percent}{\%\xspace}
\newcommand{\degrees}{\textdegree\xspace}
\providecommand{\approxeq}{$\approx$\xspace}
