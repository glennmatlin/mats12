% macros/references.tex — semantic cross-reference shortcuts.
%
% Instead of writing  Figure~\ref{fig:overview}
% you write           \reffig{overview}
%
% Benefits: enforces label naming convention, easier to globally restyle
% (e.g., switch "Figure" → "Fig." for a journal that wants short refs).

\newcommand{\reffig}[1]{Figure~\ref{fig:#1}}
\newcommand{\reftab}[1]{Table~\ref{tab:#1}}
\newcommand{\refsec}[1]{Section~\ref{sec:#1}}
\newcommand{\refeq}[1]{Equation~\ref{eq:#1}}
\newcommand{\refapp}[1]{Appendix~\ref{app:#1}}
\newcommand{\refalg}[1]{Algorithm~\ref{algo:#1}}
\newcommand{\reflst}[1]{Listing~\ref{lst:#1}}

% Compact variants (use for journals that prefer "Fig." / "Tab.")
\newcommand{\reffigshort}[1]{Fig.~\ref{fig:#1}}
\newcommand{\reftabshort}[1]{Tab.~\ref{tab:#1}}

% Ranges
\newcommand{\reffigs}[2]{Figures~\ref{fig:#1}--\ref{fig:#2}}
\newcommand{\reftabs}[2]{Tables~\ref{tab:#1}--\ref{tab:#2}}
\newcommand{\refsecs}[2]{Sections~\ref{sec:#1}--\ref{sec:#2}}
